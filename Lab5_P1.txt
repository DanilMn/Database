
CREATE DATABASE IF NOT EXISTS steam_catalog_lab
  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE steam_catalog_lab;

CREATE TABLE Developers (
  developer_id INT PRIMARY KEY AUTO_INCREMENT,
  developer_name VARCHAR(100) NOT NULL,
  country VARCHAR(50)
) ENGINE=InnoDB;

CREATE TABLE Publishers (
  publisher_id INT PRIMARY KEY AUTO_INCREMENT,
  publisher_name VARCHAR(100) NOT NULL,
  country VARCHAR(50)
) ENGINE=InnoDB;

CREATE TABLE Tags (
  tag_id INT PRIMARY KEY AUTO_INCREMENT,
  tag_name VARCHAR(100) NOT NULL UNIQUE
) ENGINE=InnoDB;

CREATE TABLE Platforms (
  platform_id INT PRIMARY KEY AUTO_INCREMENT,
  platform_name VARCHAR(50) NOT NULL UNIQUE
) ENGINE=InnoDB;

CREATE TABLE Games (
  app_id INT PRIMARY KEY,
  game_name VARCHAR(200) NOT NULL,
  developer_id INT NOT NULL,
  publisher_id INT NOT NULL,
  release_date DATE,
  is_available BOOLEAN NOT NULL,
  CONSTRAINT fk_games_developers FOREIGN KEY (developer_id)
    REFERENCES Developers(developer_id),
  CONSTRAINT fk_games_publishers FOREIGN KEY (publisher_id)
    REFERENCES Publishers(publisher_id)
) ENGINE=InnoDB;

CREATE TABLE GameTags (
  app_id INT NOT NULL,
  tag_id INT NOT NULL,
  PRIMARY KEY (app_id, tag_id),
  CONSTRAINT fk_gametags_games FOREIGN KEY (app_id) REFERENCES Games(app_id),
  CONSTRAINT fk_gametags_tags FOREIGN KEY (tag_id) REFERENCES Tags(tag_id)
) ENGINE=InnoDB;

CREATE TABLE GamePlatforms (
  app_id INT NOT NULL,
  platform_id INT NOT NULL,
  PRIMARY KEY (app_id, platform_id),
  CONSTRAINT fk_gameplatforms_games FOREIGN KEY (app_id) REFERENCES Games(app_id),
  CONSTRAINT fk_gameplatforms_platforms FOREIGN KEY (platform_id) REFERENCES Platforms(platform_id)
) ENGINE=InnoDB;

CREATE TABLE Prices (
  price_id INT PRIMARY KEY AUTO_INCREMENT,
  app_id INT NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  currency VARCHAR(10) NOT NULL,
  price_date DATE NOT NULL,
  CONSTRAINT chk_price_nonnegative CHECK (price >= 0),
  CONSTRAINT fk_prices_games FOREIGN KEY (app_id) REFERENCES Games(app_id)
) ENGINE=InnoDB;

UNION ALL SELECT 'GamePlatforms', COUNT(*) FROM GamePlatforms

SELECT g.app_id, g.game_name, d.developer_name, p.publisher_name
FROM Games AS g
JOIN Developers AS d ON g.developer_id = d.developer_id
JOIN Publishers AS p ON g.publisher_id = p.publisher_id
ORDER BY g.game_name;
